#Split AMR genes
amr_long <- isolates1 %>%
  # Keep isolates even if no AMR genes
  mutate(AMR.genotypes = if_else(
    is.na(AMR.genotypes) | AMR.genotypes == "",
    "None",
    AMR.genotypes
  )) %>%

  # Split genes into rows
  separate_rows(AMR.genotypes, sep = ",") %>%
  mutate(
    AMR_gene = str_trim(AMR.genotypes),
    AMR_gene = na_if(AMR_gene, "None")
  ) %>%
  filter(!is.na(AMR_gene)) %>%
  #Keeping only the COMPLETED profiles
  filter(str_detect(AMR_gene, "=COMPLETE")) %>%     
  mutate(AMR_gene = str_remove(AMR_gene, "=COMPLETE"))

write_csv(amr_long, here("data/processed-data/", "amr_long.csv"))
# 2. Assign antimicrobial class

amr_long <- amr_long %>%
  mutate(
    AMR_class = case_when(
      # Beta-lactams
      str_detect(AMR_gene, "^bla") ~ "Beta-lactam",
      
      # Tetracyclines
      str_detect(AMR_gene, "tet") ~ "Tetracycline",
      
      # Sulfonamides
      str_detect(AMR_gene, "^sul") ~ "Sulfonamide",
      
      # Trimethoprim (often co-resistance with sul)
      str_detect(AMR_gene, "^dfr") ~ "Trimethoprim",
      
      # Quinolones (including efflux pumps)
      str_detect(AMR_gene, "qnr|^oqx|^qepA") ~ "Quinolone",
      
      # Aminoglycosides (broadened pattern)
      str_detect(AMR_gene, "aad|aph\\(|aac\\(|armA|^rmt|^ant") ~ "Aminoglycoside",
      
      # Phenicols (chloramphenicol family)
      str_detect(AMR_gene, "floR|^cat|^cml") ~ "Phenicol",
      
      # Macrolides (broadened pattern)
      str_detect(AMR_gene, "erm|^mef|^mph|^msr|^ere") ~ "Macrolide",
      
      # Lincosamides
      str_detect(AMR_gene, "^lnu") ~ "Lincosamide",
      
      # Fosfomycin
      str_detect(AMR_gene, "^fos") ~ "Fosfomycin",
      
      # Colistin/Polymyxin
      str_detect(AMR_gene, "^mcr") ~ "Colistin",
      
      # Glycopeptides
      str_detect(AMR_gene, "^van") ~ "Glycopeptide",
      
      # Rifamycins
      str_detect(AMR_gene, "^arr") ~ "Rifamycin",

      # Multidrug efflux pumps (RND family)
      str_detect(AMR_gene, "^mds|^acr|^mex|^ade|^cme|^amrB") ~ "Efflux Pump",
      
      # Default catch-all
      TRUE ~ "Other"
    )
  )
